cmake_minimum_required(VERSION 3.21)

if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/qpm_defines.cmake")
  include("${CMAKE_CURRENT_LIST_DIR}/qpm_defines.cmake")
endif()

if(NOT DEFINED MOD_ID)
  set(MOD_ID "twitch-bsr")
endif()
if(NOT DEFINED MOD_VERSION)
  set(MOD_VERSION "0.1.0")
endif()
if(NOT DEFINED COMPILE_ID)
  set(COMPILE_ID "twitch-bsr")
endif()
if(NOT DEFINED EXTERN_DIR)
  set(EXTERN_DIR "${CMAKE_CURRENT_LIST_DIR}/extern")
endif()
if(NOT DEFINED SHARED_DIR)
  set(SHARED_DIR "${CMAKE_CURRENT_LIST_DIR}/shared")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

project(${COMPILE_ID} LANGUAGES C CXX)

add_compile_options(-frtti -fexceptions -fvisibility=default -O2)
add_compile_definitions(VERSION="${MOD_VERSION}" MOD_ID="${MOD_ID}")

file(GLOB_RECURSE cpp_files CONFIGURE_DEPENDS ${CMAKE_CURRENT_LIST_DIR}/src/*.cpp)
add_library(${COMPILE_ID} SHARED ${cpp_files})
target_include_directories(${COMPILE_ID} PRIVATE ${CMAKE_CURRENT_LIST_DIR}/include ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(${COMPILE_ID} PRIVATE log z)   # z = zlib from the Android NDK (map extraction)

# beatsaber-hook's inline-hook sources provide A64HookFunction; they must be compiled into the mod.
set(BSH_DIR "${CMAKE_CURRENT_LIST_DIR}/extern/includes/beatsaber-hook")
file(GLOB_RECURSE inline_hook_src CONFIGURE_DEPENDS
  ${BSH_DIR}/shared/inline-hook/*.c
  ${BSH_DIR}/shared/inline-hook/*.cpp
  ${BSH_DIR}/src/inline-hook/*.c
  ${BSH_DIR}/src/inline-hook/*.cpp)
if(inline_hook_src)
  message(STATUS "Adding inline-hook sources: ${inline_hook_src}")
  target_sources(${COMPILE_ID} PRIVATE ${inline_hook_src})
else()
  file(GLOB_RECURSE bsh_listing ${BSH_DIR}/*)
  message(FATAL_ERROR "No inline-hook sources found under ${BSH_DIR}. Files present: ${bsh_listing}")
endif()

include("${CMAKE_CURRENT_LIST_DIR}/extern.cmake")   # generated by `qpm restore`

# Scotland2 loader library.
file(GLOB sl2_lib "${CMAKE_CURRENT_LIST_DIR}/extern/libs/libsl2.so")
if(sl2_lib)
  message(STATUS "Linking ${sl2_lib}")
  target_link_libraries(${COMPILE_ID} PRIVATE ${sl2_lib})
else()
  message(WARNING "libsl2.so not found in extern/libs")
endif()
