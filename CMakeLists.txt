cmake_minimum_required(VERSION 3.21)
include(qpm_defines.cmake)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

project(${COMPILE_ID})
file(GLOB_RECURSE cpp_files src/*.cpp)

add_library(${COMPILE_ID} SHARED ${cpp_files})
target_include_directories(${COMPILE_ID} PRIVATE include)
target_compile_definitions(${COMPILE_ID} PRIVATE MOD_ID="${MOD_ID}" VERSION="${MOD_VERSION}" ANDROID)

include(extern.cmake)  # generated by `qpm restore`
